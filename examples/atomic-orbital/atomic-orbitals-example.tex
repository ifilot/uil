      font=\scriptsize\bfseries,inner xsep=5pt,inner ysep=2.5pt]
      at ([xshift=-2mm,yshift=-2mm]render.north east)
      {$\blacktriangleright$\ INTERACTIVE};
  \end{tikzpicture}%
}
