\newcommand{\posterbase}[1]{%
  \fill[gray!4] (0,0) rectangle (12,5.4);
  \draw[gray!45] (0,0) rectangle (12,5.4);
  \draw[gray!45] (6.15,.25) -- (6.15,5.15);
  \node[anchor=north west,font=\small\bfseries] at (.2,5.18) {#1 orbital and sampling plane};
  \node[anchor=north west,font=\small\bfseries] at (6.35,5.18) {Signed $\psi$ contour};
  \fill[white,opacity=.62] (1.0,1.0) -- (5.4,1.55) -- (5.4,4.05) -- (1.0,3.5) -- cycle;
  \draw[cyan!55!black,line width=.8pt] (1.0,1.0) -- (5.4,1.55) -- (5.4,4.05) -- (1.0,3.5) -- cycle;
  \draw[gray!55] (6.55,.65) rectangle (11.45,4.65);
  \shade[bottom color=phasered,top color=white]
    (11.08,.8) rectangle (11.28,2.65);
  \shade[bottom color=white,top color=phaseblue]
    (11.08,2.65) rectangle (11.28,4.5);
  \draw[gray!50] (11.08,.8) rectangle (11.28,4.5);
  \node[font=\tiny,anchor=west] at (11.32,4.5) {$+\max$};
  \node[font=\tiny,anchor=west] at (11.32,2.65) {$0$};
  \node[font=\tiny,anchor=west] at (11.32,.8) {$-\max$};
  \node[anchor=south east,rounded corners=1.5pt,fill=cyan!55!blue,text=white,
    font=\scriptsize\bfseries,inner xsep=5pt,inner ysep=2.5pt]
    at (11.8,.18) {$\blacktriangleright$\ INTERACTIVE};
}
