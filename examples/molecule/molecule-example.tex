% The poster remains visible in ordinary PDF readers. UIL places the live
% OpenGL molecule over the same annotation rectangle during a presentation.
\newcommand{\moleculeposter}[1]{%
  \includegraphics[width=10.8cm,height=4.2cm]{#1}%
}

\begin{document}

% The CHC theme reserves physical page 1 for its title layout. This test deck
% intentionally consists only of content slides, so start with the content
% background while retaining conventional frame numbering.
\setcounter{page}{2}

\begin{frame}{Meet the molecule toolbar}{The controls unfold from the top-right corner of every molecule}
  \centering
  \molecule[width=10.8cm,height=4.2cm]
    {\moleculeposter{methane-still.png}}
    {methane-inline.xyz}\\[0.25cm]
  \small Drag to rotate, scroll to zoom, and double-click to reset. Use the rotate
  button for a continuous spin around molecular Z. Collapse the toolbar with its
  teal close button and unfold it again with the sliders button.
\end{frame}

\begin{frame}{Stereoscopic rendering}{Toggle red/cyan anaglyph rendering with the glasses button}
  \centering
  \molecule[width=10.8cm,height=4.2cm]
    {\moleculeposter{methane-anaglyph-still.png}}
    {methane-inline.xyz}\\[0.25cm]
  \small The two stereo cameras converge on the molecule center. Press the
  glasses button once to enable red/cyan and again to return to mono rendering.
\end{frame}

\begin{frame}{Normal-mode vibrations}{Use the play button to animate the water bending mode}
  \centering
  \molecule[width=10.8cm,height=4.2cm]
    {\moleculeposter{water-still.png}}
    {water-inline.xyz}\\[0.25cm]
  \small Vibration-enabled XYZ records append $d_x$, $d_y$, and $d_z$ to each
  atom. UIL applies the displacement sinusoidally while retaining interactive rotation.
\end{frame}

\begin{frame}{Orientation axes}{Toggle the Blender-style X/Y/Z gizmo with the compass button}
  \centering
  \molecule[width=10.8cm,height=4.2cm]
    {\moleculeposter{carbon-dioxide-still.png}}
    {carbon-dioxide-inline.xyz}\\[0.25cm]
  \small Rotate the molecule and watch the corner gizmo follow it: X is red,
  Y is green, and Z is blue. This molecule also contains an asymmetric stretch.
\end{frame}

\begin{frame}{Planar molecules}{Benzene starts face-on in the YZ plane}
  \centering
  \molecule[width=10.8cm,height=4.2cm]
    {\moleculeposter{benzene-still.png}}
    {benzene-inline.xyz}\\[0.25cm]
  \small The default camera looks along $-X$ with $+Z$ upward, so the aromatic
  ring is immediately visible. Toggle continuous Z-axis rotation to inspect its
  planar geometry from different angles.
\end{frame}
