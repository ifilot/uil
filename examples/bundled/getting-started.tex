\begin{frame}{Audience controls}
  \Large
  Right-click anywhere on the audience screen to open presentation tools.
  \vspace{0.6cm}
  \begin{itemize}
    \item Press \textbf{L} for the laser pointer.
    \item Press \textbf{B} or \textbf{W} for a black or white screen.
    \item Choose pencil or eraser to work with annotations.
    \item Adjust pointer size and color in the tool panel.
  \end{itemize}
\end{frame}

\begin{frame}{Keep your work}
  \Large
  \begin{block}{Save as a uil presentation}
    Use \textbf{File $>$ Save As} to package the PDF and your annotations together in a \texttt{.uil} file.
  \end{block}
  \vspace{0.4cm}
  \begin{block}{Share an annotated PDF}
    Use \textbf{File $>$ Export as PDF} to bake visible annotations into a new PDF.
  \end{block}
\end{frame}
