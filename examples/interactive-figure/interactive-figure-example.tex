  text=white,

\newcommand{\interactivepreviewbadge}{%
  \coordinate (uilpreviewnorthwest) at (current bounding box.north west);%
  \coordinate (uilpreviewnortheast) at (current bounding box.north east);%
  % Reserve a narrow header inside the poster's measured rectangle. This keeps
  % the badge away from both the graph title and the plotted data.
  \path ([yshift=5.5mm]uilpreviewnorthwest)
    -- ([yshift=5.5mm]uilpreviewnortheast);%
  \node[uil interactive badge]
    at ([xshift=-1.5mm,yshift=4.5mm]uilpreviewnortheast) {%
      \raisebox{0.12ex}{\scalebox{0.72}{$\blacktriangleright$}}%
      \hspace{0.3em}\strut INTERACTIVE};%
}
